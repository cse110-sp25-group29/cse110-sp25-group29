        cancelBtn.addEventListener('click', ()=> {
            document.body.removeChild(deleteOverlay);

        deleteConfirmation.appendChild(confirmBtn);
        deleteOverlay.appendChild(deleteConfirmation);
        document.body.appendChild(deleteOverlay);

}
