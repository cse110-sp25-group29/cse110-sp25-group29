function rgbToHex(r, g, b) {
  const rgb = (r << 16) | (g << 8) | b;
  const str = rgb.toString(16);
  const fill = '0';
  return `#${fill.repeat(6 - str.length)}${str}`;
}

function hexToRgb(hex) {
  try {
    const rgb = parseInt(hex.substring(1), 16);
    const r = rgb >> 16;
    const g = (rgb >> 8) % 256;
    const b = rgb % 256;
    return [r, g, b];
  } catch (e) {
    return [0, 0, 0];
  }
}
